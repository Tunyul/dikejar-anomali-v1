# UID Verification & Item Granting API Contract

**Version**: 1.0
**Date**: 2026-04-12
**Purpose**: API contract between Web Top-up Server (Node.js) and Game Backend for Anomaly Rush

---

## Architecture Note

Anomaly Rush currently uses local save (`user://save.cfg`). For web top-up integration, you must implement a centralized backend that:

- Maintains player database (UID, balances, inventory)
- Syncs with game client via cloud save API
- Handles all top-up transactions securely

This document defines the API endpoints your Node.js web server will call.

---

## 1. UID Verification Endpoint

### Method & URL

```
POST /api/v1/game/verify_uid
```

### Purpose

Validate that a UID exists in the game database before allowing purchase.

### Request Headers

```json
{
  "Content-Type": "application/json",
  "X-API-Key": "your_secure_api_key"
}
```

### Request Body

```json
{
  "uid": "player_12345",
  "challenge_code": "random_string_for_verification"
}
```

**Fields:**

- `uid` (string, required): Player's unique identifier (6-20 alphanumeric characters)
- `challenge_code` (string, required): Random string generated by web server for request validation

### Response - Success (200 OK)

```json
{
  "is_valid": true,
  "player_name": "AnomalyChaser99",
  "account_status": "active",
  "level": 15
}
```

### Response - Invalid UID (404 Not Found)

```json
{
  "is_valid": false,
  "player_name": null,
  "error": "UID not found in database"
}
```

### Response - Error (500 Internal Server Error)

```json
{
  "is_valid": false,
  "player_name": null,
  "error": "Database connection failed"
}
```

### Implementation Notes

- Validate UID format: `^[a-zA-Z0-9_]{6,20}$`
- Rate limit: Max 10 requests per minute per IP
- Log all verification attempts for security audit
- Response time target: < 200ms

## 2. Item Granting Endpoint (Post-Payment Settlement)

### Method & URL

```
POST /api/v1/game/grant_item
```

### Purpose

Add purchased items/currency to player account after successful payment confirmation.

### Security

- **NEVER** expose this endpoint to client browsers
- Only call from: Payment gateway webhook OR your verified web server
- Validate webhook signatures from payment provider

### Request Headers

```json
{
  "Content-Type": "application/json",
  "X-API-Key": "your_secure_api_key",
  "X-Webhook-Signature": "sha256_signature_from_payment_gateway"
}
```

### Request Body

```json
{
  "uid": "player_12345",
  "product_id": "gems_30",
  "transaction_id": "trx_abc123def456"
}
```

**Fields:**

- `uid` (string, required): Player UID
- `product_id` (string, required): Must match product catalog ID (e.g., `gems_30`, `starter_bundle_premium`)
- `transaction_id` (string, required): Unique payment transaction ID from gateway

### Response - Success (200 OK)

```json
{
  "status": "success",
  "new_balance": {
    "gems": 65,
    "coins": 2450
  },
  "granted_at": "2026-04-12T10:15:31Z"
}
```

### Response - Idempotent (Already Processed) (200 OK)

```json
{
  "status": "success",
  "already_processed": true,
  "new_balance": {
    "gems": 65,
    "coins": 2450
  },
  "original_grant_time": "2026-04-12T10:15:31Z"
}
```

### Response - Invalid Product (400 Bad Request)

```json
{
  "status": "error",
  "error_code": "INVALID_PRODUCT",
  "message": "Product 'gems_999' not found in catalog"
}
```

### Response - Player Not Found (404 Not Found)

```json
{
  "status": "error",
  "error_code": "PLAYER_NOT_FOUND",
  "message": "UID 'player_12345' does not exist"
}
```

### Response - Server Error (500 Internal Server Error)

```json
{
  "status": "error",
  "error_code": "GRANT_FAILED",
  "message": "Failed to update player balance",
  "retry_recommended": true
}
```

## 3. Idempotency Rule

### Problem

Payment gateways may send duplicate webhooks due to network retries or timeouts. Without idempotency, players receive double rewards.

### Solution

The game backend MUST handle duplicate `transaction_id` values gracefully:

**Rule**: If `transaction_id` already exists in database with `status = 'granted'`:

1. Return HTTP 200 with `status: "success"`
2. Return `already_processed: true`
3. Do NOT grant items again
4. Return original `new_balance` from first processing

### Database Schema (PostgreSQL Example)

```sql
CREATE TABLE transactions (
  id SERIAL PRIMARY KEY,
  transaction_id VARCHAR(64) UNIQUE NOT NULL,
  uid VARCHAR(20) NOT NULL,
  product_id VARCHAR(50) NOT NULL,
  status VARCHAR(20) DEFAULT 'pending',
  granted_at TIMESTAMP,
  gems_before INTEGER,
  coins_before INTEGER,
  gems_after INTEGER,
  coins_after INTEGER,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_transaction_id ON transactions(transaction_id);
CREATE INDEX idx_uid ON transactions(uid);
```

### Idempotency Logic (Pseudocode)

```javascript
async function grantItem(uid, product_id, transaction_id) {
  // Step 1: Check if already processed
  const existing = await db.query(
    "SELECT * FROM transactions WHERE transaction_id = $1",
    [transaction_id],
  );

  if (existing.rows.length > 0 && existing.rows[0].status === "granted") {
    // Already granted - return success without re-granting
    return {
      status: "success",
      already_processed: true,
      new_balance: {
        gems: existing.rows[0].gems_after,
        coins: existing.rows[0].coins_after,
      },
      original_grant_time: existing.rows[0].granted_at,
    };
  }

  // Step 2: Start database transaction
  const client = await db.connect();

  try {
    await client.query("BEGIN");

    // Step 3: Get current player balance (with row lock)
    const player = await client.query(
      "SELECT gems, coins FROM players WHERE uid = $1 FOR UPDATE",
      [uid],
    );

    if (player.rows.length === 0) {
      throw new Error("Player not found");
    }

    // Step 4: Calculate new balance based on product
    const product = await getProductById(product_id);
    let newGems = player.rows[0].gems;
    let newCoins = player.rows[0].coins;

    if (product.grant_data.currency === "gems") {
      newGems += product.grant_data.amount;
    } else if (product.grant_data.currency === "coins") {
      newCoins += product.grant_data.amount;
    }

    // Step 5: Update player balance
    await client.query(
      "UPDATE players SET gems = $1, coins = $2 WHERE uid = $3",
      [newGems, newCoins, uid],
    );

    // Step 6: Record transaction as granted
    await client.query(
      `INSERT INTO transactions
       (transaction_id, uid, product_id, status, granted_at,
        gems_before, coins_before, gems_after, coins_after)
       VALUES ($1, $2, $3, 'granted', NOW(), $4, $5, $6, $7)`,
      [
        transaction_id,
        uid,
        product_id,
        player.rows[0].gems,
        player.rows[0].coins,
        newGems,
        newCoins,
      ],
    );

    await client.query("COMMIT");

    return {
      status: "success",
      new_balance: {
        gems: newGems,
        coins: newCoins,
      },
      granted_at: new Date().toISOString(),
    };
  } catch (error) {
    await client.query("ROLLBACK");
    console.error("Grant failed:", error);

    return {
      status: "error",
      error_code: "GRANT_FAILED",
      message: error.message,
      retry_recommended: true,
    };
  } finally {
    client.release();
  }
}
```

### Key Idempotency Principles

1. **Unique Constraint**: `transaction_id` has UNIQUE constraint in database
2. **Check Before Grant**: Always query if transaction exists before processing
3. **Atomic Operations**: Use database transactions (BEGIN/COMMIT/ROLLBACK)
4. **FOR UPDATE Lock**: Prevent race conditions when updating player balance
5. **Idempotent Response**: Return same response structure for duplicate requests
6. **Audit Trail**: Keep full transaction history with before/after balances

## 4. Webhook Signature Verification

### Midtrans Example

```javascript
const crypto = require("crypto");

function verifyMidtransSignature(payload, signature, serverKey) {
  const orderedKeys = Object.keys(payload).sort();
  const stringToSign = orderedKeys
    .map((key) => `${key}=${payload[key]}`)
    .join("&");

  const expectedSignature = crypto
    .createHash("sha512")
    .update(stringToSign + serverKey)
    .digest("hex");

  return signature === expectedSignature;
}
```

### Usage in Express Route

```javascript
app.post("/api/v1/webhook/payment", async (req, res) => {
  const signature = req.headers["x-webhook-signature"];
  const payload = req.body;

  // Verify signature
  if (!verifySignature(payload, signature, process.env.PAYMENT_SECRET)) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  // Call grant_item endpoint
  const result = await grantItem(
    payload.uid,
    payload.product_id,
    payload.transaction_id,
  );

  // Always return 200 to prevent gateway retries
  res.status(200).json(result);
});
```

## 5. Security Best Practices

### API Key Management

```javascript
// Environment variables (NEVER hardcode)
const API_KEY = process.env.WEB_TO_GAME_API_KEY;

// Middleware to validate API key
function authenticateAPI(req, res, next) {
  const apiKey = req.headers["x-api-key"];

  if (!apiKey || apiKey !== API_KEY) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  next();
}

app.use("/api/v1/game/*", authenticateAPI);
```

### Rate Limiting

```javascript
const rateLimit = require("express-rate-limit");

const verifyLimiter = rateLimit({
  windowMs: 60 * 1000, // 1 minute
  max: 10, // 10 requests per minute
  message: { error: "Too many requests" },
});

app.post("/api/v1/game/verify_uid", verifyLimiter, handleVerifyUID);
```
